```python
def convert(raw_dir, out_dir, cams=None, skip_rain=False, min_line_px=0, verbose=False):
    import os, json, cv2
    recs = []
    for f in sorted(os.listdir(raw_dir)):
        if not f.endswith("_rgb.png"): continue
        sid = f.split("_")[0]
        cam = f.split("_")[1]
        if cams and cam not in cams: continue
        meta = json.load(open(raw_dir + "/" + sid + "_meta.json"))
        if skip_rain and meta["weather"] == "rain": continue
        lab = cv2.imread(raw_dir + "/" + sid + "_" + cam + "_stencil.png", 0)
        lab2 = lab.copy()
        for k, v in {1: 3, 2: 4, 7: 5, 9: 255}.items(): lab2[lab == k] = v
        if min_line_px and (lab2 == 3).sum() < min_line_px: continue
        cv2.imwrite(out_dir + "/label/" + sid + ".png", lab2)
        os.system("cp " + raw_dir + "/" + f + " " + out_dir + "/rgb/" + sid + ".png")
        recs.append({"id": sid, "camera": cam, **meta})
        if verbose: print("ok", sid)
    with open(out_dir + "/manifest.jsonl", "w") as fh:
        for r in recs: fh.write(json.dumps(r) + "\n")
    return len(recs)
```

```python
def compute_class_weights(loader, n_cls):
    counts = torch.zeros(n_cls)
    for _, y in loader:
        y = y[y != 255]
        counts += torch.bincount(y.flatten(), minlength=n_cls)
    w = 1.0 / torch.log(1.02 + counts / counts.sum())
    return w
```

```python
def boundary_f1(preds, gts, n_cls, tol=2):
    scores = []
    for p, g in zip(preds, gts):
        for c in range(n_cls):
            pb = find_boundaries(p == c)
            gb = find_boundaries(g == c)
            if gb.sum() == 0: continue
            pd = distance_transform_edt(~pb); gd = distance_transform_edt(~gb)
            prec = (gd[pb] <= tol).mean() if pb.any() else 0
            rec = (pd[gb] <= tol).mean()
            scores.append(0 if prec + rec == 0 else 2 * prec * rec / (prec + rec))
    return float(np.mean(scores))
```
